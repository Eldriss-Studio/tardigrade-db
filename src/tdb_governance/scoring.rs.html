1//! Importance scoring (ι) for the Adaptive Knowledge Lifecycle.
2//!
3//! ι ∈ [0, 100]. Access: +3, Update: +5, Daily decay: ×0.995.
4//! Bounded to prevent unbounded accumulation.
5
6/// Daily decay factor applied to importance scores.
7const DAILY_DECAY_FACTOR: f32 = 0.995;
8
9/// Maximum importance score.
10const MAX_IMPORTANCE: f32 = 100.0;
11
12/// Minimum importance score.
13const MIN_IMPORTANCE: f32 = 0.0;
14
15/// Boost applied on each read access.
16const ACCESS_BOOST: f32 = 3.0;
17
18/// Boost applied on each update/write.
19const UPDATE_BOOST: f32 = 5.0;
20
21/// Tracks and manages the importance score for a single memory cell.
22///
23/// Uses the Observer pattern: callers notify the scorer of access/update events,
24/// and the scorer maintains the bounded ι value.
25#[derive(Debug, Clone)]
26pub struct ImportanceScorer {
27    importance: f32,
28}
29
30impl ImportanceScorer {
31    /// Create a new scorer with an initial importance value.
32    #[must_use]
33    pub fn new(initial: f32) -> Self {
34        Self { importance: initial.clamp(MIN_IMPORTANCE, MAX_IMPORTANCE) }
35    }
36
37    /// Current importance score.
38    #[must_use]
39    pub fn importance(&self) -> f32 {
40        self.importance
41    }
42
43    /// Record a read access event (+3).
44    pub fn on_access(&mut self) {
45        self.importance = (self.importance + ACCESS_BOOST).min(MAX_IMPORTANCE);
46    }
47
48    /// Record an update/write event (+5).
49    pub fn on_update(&mut self) {
50        self.importance = (self.importance + UPDATE_BOOST).min(MAX_IMPORTANCE);
51    }
52
53    /// Apply `n` days of decay (ι × 0.995^n).
54    ///
55    /// Days beyond 10,000 are clamped (the result is effectively 0 after ~4600 days).
56    ///
57    /// # Panics
58    /// Cannot panic in practice: the `i32::try_from` is on the clamped value
59    /// (≤ 10,000), which always fits in `i32`. The `.expect()` is a defensive
60    /// guard against future refactors that change the clamp constant.
61    pub fn apply_daily_decay(&mut self, days: u32) {
62        let clamped = days.min(10_000);
63        // Safe: clamped ≤ 10_000, which always fits in i32.
64        let exponent = i32::try_from(clamped).expect("clamped to 10_000");
65        self.importance = (self.importance * DAILY_DECAY_FACTOR.powi(exponent)).max(MIN_IMPORTANCE);
66    }
67
68    /// Check if this cell is below the eviction threshold.
69    #[must_use]
70    pub fn is_evictable(&self, threshold: f32) -> bool {
71        self.importance < threshold
72    }
73}

79    #[test]
80    fn test_initial_clamping() {
81        assert!((ImportanceScorer::new(150.0).importance() - 100.0).abs() < f32::EPSILON);
82        assert!((ImportanceScorer::new(-10.0).importance() - 0.0).abs() < f32::EPSILON);
83    }

86    fn test_access_boost() {
87        let mut s = ImportanceScorer::new(0.0);
88        s.on_access();
89        assert!((s.importance() - 3.0).abs() < f32::EPSILON);
90    }
91
92    #[test]
93    fn test_update_boost() {
94        let mut s = ImportanceScorer::new(0.0);
95        s.on_update();
96        assert!((s.importance() - 5.0).abs() < f32::EPSILON);
97    }

99    #[test]
100    fn test_max_cap() {
101        let mut s = ImportanceScorer::new(99.0);
102        s.on_update(); // 99 + 5 = 104 → capped at 100
103        assert!((s.importance() - 100.0).abs() < f32::EPSILON);
104    }

108        let mut s = ImportanceScorer::new(100.0);

120    /// Numerical-precision AT for [`apply_daily_decay`].
121    ///
122    /// Guards the policy decision (workspace `Cargo.toml` `[lints]`) to allow
123    /// `cast_precision_loss` workspace-wide: the f32 decay accumulator must
124    /// track the analytical formula `ι₀ × DAILY_DECAY_FACTOR^n` to within
125    /// tier-relevant precision (the smallest hysteresis band is 5 importance
126    /// points). If a future refactor switches to a numerically-unstable
127    /// formulation, this AT fails before any production deployment regression.
128    ///
129    /// The reference value is the formula itself, computed in f64 — the
130    /// production path is f32, so any divergence beyond rounding is a real
131    /// numerical regression. Sampled across the full lifetime of an
132    /// importance score (day 0 to the 10 000-day clamp).
133    #[test]
134    fn daily_decay_tracks_analytical_formula() {
135        /// Reference: the spec for `apply_daily_decay` is exactly
136        /// `ι₀ × DAILY_DECAY_FACTOR^n`. Compute in f64 so the test
137        /// captures the analytical truth, not a transcribed snapshot.
138        fn analytical_decay(initial: f64, days: u32) -> f64 {
139            initial * f64::from(DAILY_DECAY_FACTOR).powi(i32::try_from(days).unwrap())
140        }
141
142        let initial = f64::from(MAX_IMPORTANCE);
143        let days_to_check: &[u32] = &[0, 1, 7, 30, 90, 365, 1_000, 5_000];
144
145        for &days in days_to_check {
146            let expected = analytical_decay(initial, days);
147            let mut s = ImportanceScorer::new(MAX_IMPORTANCE);
148            s.apply_daily_decay(days);
149            let got = f64::from(s.importance());
150            // Tier hysteresis bands are ≥ 5 points; we hold a tighter
151            // bar so the test fails on real numerical drift, not noise.
152            let tolerance = (expected.abs() * 1e-3).max(1e-6);
153            assert!(
154                (got - expected).abs() < tolerance,
155                "decay at day {days}: expected {expected:.6e}, got {got:.6e} \
156                 (tolerance {tolerance:.6e})",
157            );
158        }
159
160        // Saturation at the 10 000-day clamp: anything past that floor
161        // must stay equal (the clamp is the contract).
162        let mut clamped = ImportanceScorer::new(100.0);
163        clamped.apply_daily_decay(10_000);
164        let mut past_clamp = ImportanceScorer::new(100.0);
165        past_clamp.apply_daily_decay(50_000);
166        assert!(
167            (clamped.importance() - past_clamp.importance()).abs() < f32::EPSILON,
168            "decay must saturate at the 10 000-day clamp; got {} vs {}",
169            clamped.importance(),
170            past_clamp.importance(),
171        );
172    }
