1//! Durability boundary tracking — closes the rule-vs-reality gap that
2//! CLAUDE.md's "Reliability & Consistency Rules" left open.
3//!
4//! Two monotonic counters express the boundary:
5//!
6//! - `issued`: bumps every time a write is accepted (regardless of
7//!   whether the underlying storage has fsynced it). A write goes
8//!   from "issued" to "durable" via an fsync somewhere downstream.
9//! - `durable`: bumps every time an fsync completes successfully.
10//!   Always `<= issued`. Catches up to `issued` after `flush_buffer`
11//!   on the streaming-ingest path or after each immediate-fsync
12//!   write on the non-buffered path.
13//!
14//! Confirmed reads snapshot `issued` at request entry, run the
15//! retrieval pipeline, then wait until `durable` reaches the
16//! snapshot — every concurrent in-flight write that existed at
17//! request time is now durable when the read returns.
18//!
19//! Implementation: `std::sync::Condvar` + `AtomicU64`, no async
20//! runtime. The project's existing concurrency model is `Arc<RwLock<>>`
21//! from std accessed under `py.detach()`; staying inside std avoids
22//! dragging tokio into a synchronous codebase just for one watch
23//! channel.
24//!
25//! # Invariants
26//!
27//! - `issued` and `durable` are both monotonically non-decreasing.
28//! - `durable <= issued` at all times.
29//! - `publish_durable(n)` is idempotent: republishing an offset at
30//!   or below the current value is a no-op.
31
32use std::sync::atomic::{AtomicU64, Ordering};
33use std::sync::{Condvar, Mutex};
34use std::time::{Duration, Instant};
35
36/// Tracks the durability boundary and lets callers wait on it.
37///
38/// One [`DurabilityTracker`] per engine. Cloneable references via
39/// [`std::sync::Arc`] are how readers wait without holding the engine
40/// lock — the Python binding's confirmed-read path borrows the
41/// tracker via [`crate::engine::Engine::durability_tracker`], drops
42/// the engine read lock, then calls [`Self::wait_durable`] outside
43/// the lock so writers can advance the offset concurrently.
44#[derive(Debug, Default)]
45pub struct DurabilityTracker {
46    issued: AtomicU64,
47    durable: AtomicU64,
48    /// Companion to [`Self::notify`]. The Condvar protocol requires
49    /// holding this mutex when both signalling and waiting.
50    notify_lock: Mutex<()>,
51    notify: Condvar,
52}
53
54/// Error returned by [`DurabilityTracker::wait_durable`] when the
55/// caller's deadline passes before `durable` reaches the target.
56#[derive(Debug, Clone, Copy)]
57pub struct WaitTimedOut {
58    /// Elapsed wall-clock time the caller spent waiting before
59    /// giving up. Useful for surfacing in error messages.
60    pub waited: Duration,
61}
62
63impl DurabilityTracker {
64    /// Construct an empty tracker. Both offsets start at 0.
65    #[must_use]
66    pub fn new() -> Self {
67        Self {
68            issued: AtomicU64::new(0),
69            durable: AtomicU64::new(0),
70            notify_lock: Mutex::new(()),
71            notify: Condvar::new(),
72        }
73    }
74
75    /// Current durable offset — every write at or below this offset
76    /// has reached stable storage.
77    #[must_use]
78    pub fn current_durable(&self) -> u64 {
79        self.durable.load(Ordering::Acquire)
80    }
81
82    /// Current issued offset — every accepted write up to and
83    /// including this offset is in flight or durable.
84    #[must_use]
85    pub fn current_issued(&self) -> u64 {
86        self.issued.load(Ordering::Acquire)
87    }
88
89    /// Atomically bump `issued` and return the new value.
90    ///
91    /// Called from every write path the moment the engine accepts
92    /// the write. Distinct from durability — the write may sit in a
93    /// buffer for an arbitrary time before its fsync lands.
94    pub fn issue_one(&self) -> u64 {
95        self.issued.fetch_add(1, Ordering::AcqRel) + 1
96    }
97
98    /// Mark every issued offset up to and including `offset` as
99    /// durable. Idempotent: a second call with the same or lower
100    /// value is a no-op. Wakes every [`wait_durable`] caller.
101    ///
102    /// # Panics
103    ///
104    /// Panics if the internal notify mutex has been poisoned (a
105    /// thread held the lock and panicked). This indicates a bug
106    /// elsewhere in the engine; the durability tracker treats it as
107    /// non-recoverable.
108    ///
109    /// [`wait_durable`]: Self::wait_durable
110    pub fn publish_durable(&self, offset: u64) {
111        // `fetch_max` semantics via CAS — keep the maximum so
112        // out-of-order publishes don't regress the offset.
113        let mut current = self.durable.load(Ordering::Acquire);
114        loop {
115            if offset <= current {
116                return;
117            }
118            match self.durable.compare_exchange_weak(
119                current,
120                offset,
121                Ordering::AcqRel,
122                Ordering::Acquire,
123            ) {
124                Ok(_) => break,
125                Err(actual) => current = actual,
126            }
127        }
128        // Acquiring the notify mutex here means waiters that
129        // observed `durable < target` and are about to wait have
130        // either already entered `wait_timeout` (and will be woken
131        // by `notify_all`) or are still holding the mutex (in which
132        // case we serialise behind them and they see the updated
133        // value on their next check).
134        let _guard = self.notify_lock.lock().expect("durability notify mutex poisoned");
135        self.notify.notify_all();
136    }
137
138    /// Block until `current_durable() >= target` or `deadline` elapses.
139    ///
140    /// Returns `Ok(())` on success and `Err(WaitTimedOut { waited })`
141    /// when the deadline passes. The mutex held during waiting is
142    /// independent of the engine's `RwLock` — callers should drop
143    /// the engine lock before invoking this so writers can advance
144    /// the offset.
145    ///
146    /// # Errors
147    ///
148    /// Returns [`WaitTimedOut`] if `current_durable()` does not reach
149    /// `target` within `deadline`.
150    ///
151    /// # Panics
152    ///
153    /// Panics if the internal notify mutex has been poisoned. The
154    /// durability tracker treats poisoning as non-recoverable —
155    /// engine-level invariants are broken if a thread panicked while
156    /// holding it.
157    pub fn wait_durable(&self, target: u64, deadline: Duration) -> Result<(), WaitTimedOut> {
158        // Fast path — already durable, skip the mutex acquire.
159        if self.current_durable() >= target {
160            return Ok(());
161        }
162        let start = Instant::now();
163        let mut guard = self.notify_lock.lock().expect("durability notify mutex poisoned");
164        loop {
165            if self.current_durable() >= target {
166                return Ok(());
167            }
168            let elapsed = start.elapsed();
169            // `checked_sub` rather than `-`: clippy's strict mode
170            // forbids the unguarded operator on `Duration`.
171            let Some(remaining) = deadline.checked_sub(elapsed) else {
172                return Err(WaitTimedOut { waited: elapsed });
173            };
174            let (next_guard, timeout_result) = self
175                .notify
176                .wait_timeout(guard, remaining)
177                .expect("durability notify mutex poisoned");
178            guard = next_guard;
179            if timeout_result.timed_out() && self.current_durable() < target {
180                return Err(WaitTimedOut { waited: start.elapsed() });
181            }
182        }
183    }
184}

186#[cfg(test)]
187mod tests {
188    use super::*;
189    use std::sync::Arc;
190    use std::thread;
191
192    #[test]
193    fn new_tracker_has_zero_offsets() {
194        let t = DurabilityTracker::new();
195        assert_eq!(t.current_durable(), 0);
196        assert_eq!(t.current_issued(), 0);
197    }
198
199    #[test]
200    fn issue_one_returns_strictly_increasing_offsets() {
201        let t = DurabilityTracker::new();
202        assert_eq!(t.issue_one(), 1);
203        assert_eq!(t.issue_one(), 2);
204        assert_eq!(t.issue_one(), 3);
205        assert_eq!(t.current_issued(), 3);
206    }
207
208    #[test]
209    fn publish_durable_is_monotonic() {
210        let t = DurabilityTracker::new();
211        t.publish_durable(5);
212        assert_eq!(t.current_durable(), 5);
213        t.publish_durable(3); // Lower — must not regress.
214        assert_eq!(t.current_durable(), 5);
215        t.publish_durable(10);
216        assert_eq!(t.current_durable(), 10);
217    }
218
219    #[test]
220    fn wait_durable_returns_immediately_when_already_at_target() {
221        let t = DurabilityTracker::new();
222        t.publish_durable(5);
223        let start = Instant::now();
224        assert!(t.wait_durable(3, Duration::from_mins(1)).is_ok());
225        assert!(start.elapsed() < Duration::from_millis(10));
226    }
227
228    #[test]
229    fn wait_durable_times_out_when_no_publish_occurs() {
230        let t = DurabilityTracker::new();
231        let err = t.wait_durable(1, Duration::from_millis(50)).unwrap_err();
232        assert!(err.waited >= Duration::from_millis(50));
233        assert!(err.waited < Duration::from_millis(500));
234    }
235
236    #[test]
237    fn wait_durable_unblocks_when_publish_arrives() {
238        let t = Arc::new(DurabilityTracker::new());
239        let waiter = {
240            let t = Arc::clone(&t);
241            thread::spawn(move || t.wait_durable(5, Duration::from_secs(2)))
242        };
243        // Give the waiter a moment to enter its wait.
244        thread::sleep(Duration::from_millis(50));
245        t.publish_durable(5);
246        let result = waiter.join().unwrap();
247        assert!(result.is_ok());
248    }
249
250    #[test]
251    fn multiple_waiters_unblock_on_single_publish() {
252        let t = Arc::new(DurabilityTracker::new());
253        let waiters: Vec<_> = (0..3)
254            .map(|_| {
255                let t = Arc::clone(&t);
256                thread::spawn(move || t.wait_durable(7, Duration::from_secs(2)))
257            })
258            .collect();
259        thread::sleep(Duration::from_millis(50));
260        t.publish_durable(7);
261        for waiter in waiters {
262            assert!(waiter.join().unwrap().is_ok());
263        }
264    }
265}
