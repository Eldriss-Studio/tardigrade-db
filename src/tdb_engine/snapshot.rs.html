1//! Portable snapshot/restore for the engine.
2//!
3//! Bundles the engine's persistent state into a single tar archive
4//! with a forensic-friendly manifest header. The format is adapted
5//! from `SpacetimeDB`'s `crates/snapshot` (magic bytes + version +
6//! offset-indexed lookup) with `TardigradeDB`-specific additions
7//! (codec versioning so snapshots survive future compressor swaps).
8//!
9//! ## On-disk layout
10//!
11//! ```text
12//! <out_path>.tar
13//!   manifest.json           — versioned manifest (see SnapshotManifest)
14//!   engine_state/           — verbatim copy of the engine's working dir
15//!     <block-pool segments, WAL, text store, deletion log, ...>
16//! ```
17//!
18//! ## Reliability contract
19//!
20//! - **Durability.** [`crate::engine::Engine::snapshot`] calls [`crate::engine::Engine::flush`] before
21//!   tarring; the archive therefore reflects state at the
22//!   `durable_offset` of flush completion.
23//! - **Recovery.** [`crate::engine::Engine::restore_from`] is equivalent to a fresh
24//!   [`crate::engine::Engine::open`] of the same directory after extraction. All
25//!   existing replay rules apply unchanged.
26//! - **Versioning.** Restore refuses unknown `format_version` and
27//!   incompatible `codecs` with structured errors. Forward-compat is
28//!   an explicit conversion tool, not implicit.
29//! - **Integrity.** A SHA-256 of the `engine_state/` payload is
30//!   stored in the manifest; restore verifies it before extracting.
31

72/// Upper bound (bytes) on the manifest size during restore. The
73/// real manifest is ~500 bytes; 1 MiB is a generous ceiling that

78/// Upper bound (bytes) on a single payload file during restore.
79/// 16 GiB accommodates the largest plausible `BlockPool` segment
80/// (`segment_size` is 256 MiB by default; we leave 64× headroom for
81/// future configurations) while still bounding worst-case RAM use.
82const MAX_PAYLOAD_FILE_BYTES: u64 = 16 * 1024 * 1024 * 1024;

101/// Snapshot schema, version 1.
102///
103/// Every snapshot the engine produces today carries
104/// `format_version: 1` and decodes into the structs defined here.
105/// The module is intentionally self-contained — when v2 lands as a
106/// sibling module, v1 stays in the codebase so old snapshots
107/// produced by older engines remain restorable forward in time.
108///
109/// The on-disk contract this module pins:
110/// - Manifest JSON layout (field names, types, ordering)
111/// - `quantization = "q4"`, `key = "top5avg"` codec identifiers
112/// - SHA-256 of the `engine_state/` payload as the integrity check
113pub mod v1 {
114    use serde::{Deserialize, Serialize};
115
116    /// Format version constant for v1 snapshots.
117    pub const SNAPSHOT_FORMAT_VERSION: u8 = 1;
118
119    /// Quantization codec identifier embedded in v1 manifests.
120    /// Snapshots embed this so a future engine that drops Q4 support
121    /// can refuse Q4 snapshots cleanly instead of producing garbage
122    /// reads.
123    pub const SNAPSHOT_QUANT_CODEC: &str = "q4";
124
125    /// Per-token retrieval-key codec identifier for v1 manifests.
126    pub const SNAPSHOT_KEY_CODEC: &str = "top5avg";
127
128    /// Versioned metadata header written at the root of every v1
129    /// snapshot archive. Read first by
130    /// [`crate::engine::Engine::restore_from`] to validate
131    /// compatibility before extracting the payload.
132    #[derive(Debug, Clone, Serialize, Deserialize)]
133    pub struct SnapshotManifest {
134        /// Magic marker; must equal [`super::SNAPSHOT_MAGIC`].
135        pub magic: String,
136        /// Format version; restore refuses unknown values.
137        pub format_version: u8,
138        /// RFC 3339 timestamp of snapshot creation.
139        pub created_at: String,
140        /// Codec identifiers for the storage formats embedded in the
141        /// payload. Restore refuses snapshots whose codecs are not
142        /// supported by the current engine.
143        pub codecs: SnapshotCodecs,
144        /// Cheap stats useful for forensics + repository lookup.
145        pub stats: SnapshotStats,
146        /// SHA-256 (hex-encoded) of the tarred ``engine_state/``
147        /// payload. Computed *over the engine state files only* —
148        /// not over the manifest itself. Verified by
149        /// [`crate::engine::Engine::restore_from`].
150        pub sha256: String,
151    }
152
153    /// Codec identifiers embedded in a v1 snapshot manifest.
154    #[derive(Debug, Clone, Serialize, Deserialize)]
155    pub struct SnapshotCodecs {
156        /// Quantization codec (e.g. ``"q4"``).
157        pub quantization: String,
158        /// Per-token retrieval-key codec (e.g. ``"top5avg"``).
159        pub key: String,
160    }
161
162    /// V1 snapshot stats — handy for the repository's `latest()`
163    /// lookup and for displaying in admin UIs without unpacking the
164    /// archive.
165    #[derive(Debug, Clone, Default, Serialize, Deserialize)]
166    pub struct SnapshotStats {
167        /// Total packs in the engine at snapshot time.
168        pub pack_count: usize,
169        /// Distinct owners with at least one pack.
170        pub owner_count: usize,
171    }
172
173    impl SnapshotManifest {
174        /// Construct a v1 manifest. `sha256` and `stats` are caller-
175        /// provided; the rest come from v1's format constants.
176        #[must_use]
177        pub fn current(sha256: String, stats: SnapshotStats) -> Self {
178            Self {
179                magic: super::SNAPSHOT_MAGIC.to_string(),
180                format_version: SNAPSHOT_FORMAT_VERSION,
181                created_at: super::rfc3339_now(),
182                codecs: SnapshotCodecs {
183                    quantization: SNAPSHOT_QUANT_CODEC.to_string(),
184                    key: SNAPSHOT_KEY_CODEC.to_string(),
185                },
186                stats,
187                sha256,
188            }
189        }
190    }
191}

193/// Format an RFC 3339 timestamp without pulling in `chrono`. Returns
194/// seconds since the Unix epoch as a string; sufficient for forensic
195/// ordering and human readability inside admin tooling.
196fn rfc3339_now() -> String {
197    let secs = std::time::SystemTime::now()
198        .duration_since(std::time::UNIX_EPOCH)
199        .map_or(0, |d| d.as_secs());
200    // Compact ISO-ish: ``1970-01-01T00:00:00Z + N seconds``. We do
201    // not need calendar conversion at this layer; downstream tooling
202    // can pretty-print if needed.
203    format!("@{secs}")
204}

208/// Bundle the engine's working directory into a tar archive at
209/// ``out_path``. Returns the manifest by value so callers can log or
210/// persist it without re-reading the archive.
211///
212/// The caller must have already flushed the engine. [`crate::engine::Engine::snapshot`]
213/// in `engine.rs` is the proper public entry point and handles flush.
214///
215/// # Errors
216///
217/// Returns [`TardigradeError::Io`] on disk failures and a
218/// [`TardigradeError::SnapshotIntegrity`] if the manifest cannot be
219/// serialized (which would indicate a bug, not user input).
220pub fn write_snapshot(
221    engine_dir: &Path,
222    out_path: &Path,
223    stats: SnapshotStats,
224) -> Result<SnapshotManifest> {
225    // Refuse the foot-gun where ``out_path`` sits inside
226    // ``engine_dir`` — the tar walker would read its own growing
227    // output and exhaust disk. Caller must write the snapshot
228    // somewhere outside the engine state.
229    if let (Ok(out_abs), Ok(engine_abs)) = (
230        out_path.canonicalize().or_else(|_| {
231            // The output usually doesn't exist yet; resolve its parent.
232            out_path.parent().map_or_else(
233                || Ok(out_path.to_path_buf()),
234                |p| p.canonicalize().map(|abs| abs.join(out_path.file_name().unwrap_or_default())),
235            )
236        }),
237        engine_dir.canonicalize(),
238    ) && out_abs.starts_with(&engine_abs)
239    {
240        return Err(TardigradeError::SnapshotIntegrity(format!(
241            "out_path ({}) is inside engine_dir ({}); pick a path outside the engine directory",
242            out_path.display(),
243            engine_dir.display(),
244        )));
245    }
246
247    let mut hasher = Sha256::new();
248
249    // Step 1: hash the payload first so we can include the digest
250    // in the manifest. We walk the same set of files we'll
251    // subsequently tar so the digest matches what restore will see.
252    walk_files_for_hash(engine_dir, &mut hasher)?;
253    let sha256 = hex_lower(&hasher.finalize());
254
255    let manifest = SnapshotManifest::current(sha256, stats);
256    let manifest_json = serde_json::to_vec_pretty(&manifest)
257        .map_err(|e| TardigradeError::SnapshotIntegrity(format!("manifest serialize: {e}")))?;
258
259    // Step 2: write the tar archive: manifest.json first, then the
260    // engine_state/ payload.
261    let file = File::create(out_path).map_err(|e| TardigradeError::Io { source: e })?;
262    let mut tar = Builder::new(file);
263    tar.mode(tar::HeaderMode::Deterministic);
264
265    // Manifest entry.
266    let mut header = tar::Header::new_gnu();
267    header.set_path(SNAPSHOT_MANIFEST_NAME).map_err(|e| TardigradeError::Io { source: e })?;
268    header.set_size(manifest_json.len() as u64);
269    header.set_mode(0o644);
270    header.set_cksum();
271    tar.append(&header, &manifest_json[..]).map_err(|e| TardigradeError::Io { source: e })?;
272
273    // Payload: copy every regular file under engine_dir into
274    // engine_state/ inside the archive.
275    tar.append_dir_all(SNAPSHOT_PAYLOAD_DIR, engine_dir)
276        .map_err(|e| TardigradeError::Io { source: e })?;
277
278    tar.finish().map_err(|e| TardigradeError::Io { source: e })?;
279    Ok(manifest)
280}

282/// Read just the manifest from a snapshot archive without
283/// extracting the payload or verifying the integrity hash.
284///
285/// Useful for listing operations that only need metadata (label
286/// directories, repository enumeration). Magic and format version
287/// are still validated — a tar that lies about being a
288/// `TardigradeDB` snapshot is refused here too.
289///
290/// # Errors
291///
292/// - [`TardigradeError::NotATardigradeSnapshot`] — magic missing.
293/// - [`TardigradeError::UnsupportedFormatVersion`] — unknown version.
294/// - [`TardigradeError::Io`] — disk failure.
295pub fn read_manifest_only(in_path: &Path) -> Result<SnapshotManifest> {
296    let file = File::open(in_path).map_err(|e| TardigradeError::Io { source: e })?;
297    let mut archive = Archive::new(file);
298    for entry in archive.entries().map_err(|e| TardigradeError::Io { source: e })? {
299        let mut entry = entry.map_err(|e| TardigradeError::Io { source: e })?;
300        let path = entry.path().map_err(|e| TardigradeError::Io { source: e })?.to_path_buf();
301        if path.as_os_str() != SNAPSHOT_MANIFEST_NAME {
302            continue;
303        }
304        let buf = read_entry_bounded(&mut entry, MAX_MANIFEST_BYTES)?;
305        let manifest: SnapshotManifest = serde_json::from_slice(&buf).map_err(|e| {
306            TardigradeError::NotATardigradeSnapshot { reason: format!("manifest parse: {e}") }
307        })?;
308        if manifest.magic != SNAPSHOT_MAGIC {
309            return Err(TardigradeError::NotATardigradeSnapshot {
310                reason: format!("unexpected magic: {:?}", manifest.magic),
311            });
312        }
313        if manifest.format_version != SNAPSHOT_FORMAT_VERSION {
314            return Err(TardigradeError::UnsupportedFormatVersion {
315                found: manifest.format_version,
316                supported: SNAPSHOT_FORMAT_VERSION,
317            });
318        }
319        return Ok(manifest);
320    }
321    Err(TardigradeError::NotATardigradeSnapshot { reason: "manifest.json missing".to_string() })
322}

324/// Restore a snapshot archive into ``target_dir`` (which must be
325/// empty or non-existent) and validate its manifest. The caller
326/// invokes [`crate::engine::Engine::open`] on ``target_dir`` afterwards.
327///
328/// Returns the parsed manifest so callers can inspect or log the
329/// snapshot's metadata.
330///
331/// # Errors
332///
333/// - [`TardigradeError::NotATardigradeSnapshot`] — magic missing.
334/// - [`TardigradeError::UnsupportedFormatVersion`] — unknown version.
335/// - [`TardigradeError::SnapshotCodecMismatch`] — codec incompatible.
336/// - [`TardigradeError::SnapshotIntegrity`] — SHA-256 mismatch.
337/// - [`TardigradeError::Io`] — disk failure.
338pub fn read_snapshot(in_path: &Path, target_dir: &Path) -> Result<SnapshotManifest> {
339    let file = File::open(in_path).map_err(|e| TardigradeError::Io { source: e })?;
340    let mut archive = Archive::new(file);
341
342    // First entry: manifest. Pull it out independently so we can
343    // refuse the snapshot before extracting any payload.
344    let mut manifest_bytes: Option<Vec<u8>> = None;
345    let mut payload_entries: Vec<(PathBuf, Vec<u8>)> = Vec::new();
346
347    for entry in archive.entries().map_err(|e| TardigradeError::Io { source: e })? {
348        let mut entry = entry.map_err(|e| TardigradeError::Io { source: e })?;
349        let path = entry.path().map_err(|e| TardigradeError::Io { source: e })?.to_path_buf();
350
351        if path.as_os_str() == SNAPSHOT_MANIFEST_NAME {
352            let buf = read_entry_bounded(&mut entry, MAX_MANIFEST_BYTES)?;
353            manifest_bytes = Some(buf);
354            continue;
355        }
356        // Only regular files inside engine_state/ are part of the
357        // payload; tar Directory entries are skipped (we re-create
358        // dirs on extract).
359        if !entry.header().entry_type().is_file() {
360            continue;
361        }
362        if !path.starts_with(SNAPSHOT_PAYLOAD_DIR) {
363            continue;
364        }
365        let buf = read_entry_bounded(&mut entry, MAX_PAYLOAD_FILE_BYTES)?;
366        payload_entries.push((path, buf));
367    }
368
369    let manifest_bytes = manifest_bytes.ok_or_else(|| TardigradeError::NotATardigradeSnapshot {
370        reason: "manifest.json missing".to_string(),
371    })?;
372    let manifest: SnapshotManifest = serde_json::from_slice(&manifest_bytes).map_err(|e| {
373        TardigradeError::NotATardigradeSnapshot { reason: format!("manifest parse: {e}") }
374    })?;
375
376    if manifest.magic != SNAPSHOT_MAGIC {
377        return Err(TardigradeError::NotATardigradeSnapshot {
378            reason: format!("unexpected magic: {:?}", manifest.magic),
379        });
380    }
381    if manifest.format_version != SNAPSHOT_FORMAT_VERSION {
382        return Err(TardigradeError::UnsupportedFormatVersion {
383            found: manifest.format_version,
384            supported: SNAPSHOT_FORMAT_VERSION,
385        });
386    }
387    if manifest.codecs.quantization != SNAPSHOT_QUANT_CODEC {
388        return Err(TardigradeError::SnapshotCodecMismatch {
389            field: "quantization".to_string(),
390            snapshot: manifest.codecs.quantization.clone(),
391            engine: SNAPSHOT_QUANT_CODEC.to_string(),
392        });
393    }
394    if manifest.codecs.key != SNAPSHOT_KEY_CODEC {
395        return Err(TardigradeError::SnapshotCodecMismatch {
396            field: "key".to_string(),
397            snapshot: manifest.codecs.key.clone(),
398            engine: SNAPSHOT_KEY_CODEC.to_string(),
399        });
400    }
401
402    // Verify the payload SHA-256 against the manifest. Both write
403    // and read sides hash in *lexicographic* order of the relative
404    // path so the digest is order-independent of the underlying
405    // filesystem (OS readdir order leaked through tar entries
406    // otherwise and produced spurious mismatches).
407    payload_entries.sort_by(|a, b| a.0.cmp(&b.0));
408    let mut hasher = Sha256::new();
409    for (path, bytes) in &payload_entries {
410        let rel = strip_payload_prefix(path);
411        hash_one(&mut hasher, rel.as_os_str(), bytes);
412    }
413    let computed = hex_lower(&hasher.finalize());
414    if computed != manifest.sha256 {
415        return Err(TardigradeError::SnapshotIntegrity(format!(
416            "sha256 mismatch: manifest={}, computed={}",
417            manifest.sha256, computed
418        )));
419    }
420
421    // Extract payload into target_dir.
422    fs::create_dir_all(target_dir).map_err(|e| TardigradeError::Io { source: e })?;
423    for (path, bytes) in payload_entries {
424        let rel = strip_payload_prefix(&path);
425        let dest = target_dir.join(rel);
426        if let Some(parent) = dest.parent() {
427            fs::create_dir_all(parent).map_err(|e| TardigradeError::Io { source: e })?;
428        }
429        let mut f = File::create(&dest).map_err(|e| TardigradeError::Io { source: e })?;
430        f.write_all(&bytes).map_err(|e| TardigradeError::Io { source: e })?;
431    }
432
433    Ok(manifest)
434}

438/// Walk every regular file under ``engine_dir`` in a stable order
439/// (lexicographic by relative path) and feed each path + contents
440/// into ``hasher``. The order must match the equivalent walk on the
441/// read side, otherwise the digest will mismatch.
442fn walk_files_for_hash(engine_dir: &Path, hasher: &mut Sha256) -> Result<()> {
443    let mut entries: Vec<PathBuf> = Vec::new();
444    collect_files(engine_dir, engine_dir, &mut entries)?;
445    entries.sort();
446    for rel in entries {
447        let abs = engine_dir.join(&rel);
448        let bytes = fs::read(&abs).map_err(|e| TardigradeError::Io { source: e })?;
449        hash_one(hasher, rel.as_os_str(), &bytes);
450    }
451    Ok(())
452}
453
454fn collect_files(root: &Path, dir: &Path, out: &mut Vec<PathBuf>) -> Result<()> {
455    for entry in fs::read_dir(dir).map_err(|e| TardigradeError::Io { source: e })? {
456        let entry = entry.map_err(|e| TardigradeError::Io { source: e })?;
457        let path = entry.path();
458        if path.is_dir() {
459            collect_files(root, &path, out)?;
460        } else if path.is_file() {
461            let rel = path.strip_prefix(root).unwrap_or(&path).to_path_buf();
462            out.push(rel);
463        }
464    }
465    Ok(())
466}

472/// Read up to `max_bytes` from `r` into a fresh `Vec<u8>`, erroring
473/// if more bytes are available. Provides the [`Read::read_to_end`]
474/// convenience without the OOM exposure on corrupt archives that
475/// declare absurdly large entry sizes.
476fn read_entry_bounded<R: Read>(r: &mut R, max_bytes: u64) -> Result<Vec<u8>> {
477    // Drain the entry into a Vec via a fixed scratch buffer; refuses
478    // (with a typed integrity error) any entry that exceeds
479    // `max_bytes`. Hand-rolled rather than `Read::read_to_end` so we
480    // never allocate beyond the cap on a corrupt or hostile archive.
481    const CHUNK: usize = 64 * 1024;
482    let mut buf: Vec<u8> = Vec::new();
483    let mut scratch = vec![0u8; CHUNK];
484    // Reason: callers pass `max_bytes` as a hard cap from snapshot constants
485    // (≤ 16 GiB). On 64-bit usize the cast is lossless; on 32-bit a value
486    // above usize::MAX clamps to usize::MAX, which still upholds the cap
487    // semantics (the bound just becomes the platform's address space limit).
488    #[allow(clippy::cast_possible_truncation)]
489    let max = max_bytes as usize;
490    loop {
491        let n = r.read(&mut scratch).map_err(|e| TardigradeError::Io { source: e })?;
492        if n == 0 {
493            return Ok(buf);
494        }
495        if buf.len().saturating_add(n) > max {
496            return Err(TardigradeError::SnapshotIntegrity(format!(
497                "entry exceeds maximum allowed size of {max_bytes} bytes",
498            )));
499        }
500        buf.extend_from_slice(&scratch[..n]);
501    }
502}

504fn hash_one(hasher: &mut Sha256, name: &std::ffi::OsStr, bytes: &[u8]) {
505    // Mix the path into the hash so reordered renames cannot collide
506    // with each other.
507    hasher.update(name.as_encoded_bytes());
508    hasher.update(b"\0");
509    hasher.update((bytes.len() as u64).to_le_bytes());
510    hasher.update(bytes);
511}

513fn hex_lower(bytes: &[u8]) -> String {
514    use std::fmt::Write;
515    let mut s = String::with_capacity(bytes.len() * 2);
516    for b in bytes {
517        let _ = write!(s, "{b:02x}");
518    }
519    s
