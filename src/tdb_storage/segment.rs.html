1//! Segment file management — append-only binary files containing serialized memory cells.
2//!
3//! Each segment file has a small header followed by variable-length records.
4//! New segments are created when the current segment exceeds the size threshold.
5//!
6//! ## File Header
7//!
8//! ```text
9//! [magic: "TDBS" (4 bytes)]
10//! [version: u32]
11//! ```
12//!
13//! Two versions are supported on read:
14//!
15//! - **v1** — legacy format with no per-record codec byte. All cells are read as
16//!   [`CompressionCodec::UniformQ4`].
17//! - **v2** — adds a `codec_id` byte to each record so cells can be encoded by
18//!   different codecs in the same segment (tier-gated on write).
19//!
20//! New segments are always written in **v2**. v1 segments remain readable for
21//! backward compatibility.
22//!
23//! ## Record Layout (v2, binary, little-endian)
24//!
25//! ```text
26//! [record_len: u32]       — total bytes of this record (excluding this field)
27//! [cell_id: u64]          — globally unique cell ID
28//! [owner: u64]            — owner/agent ID
29//! [layer: u16]            — transformer layer index
30//! [key_dim: u32]          — number of f32 elements in key vector
31//! [value_dim: u32]        — number of f32 elements in value vector
32//! [pos_dim: u32]          — number of f32 elements in pos_encoding
33//! [token_span_start: u64] — token span start
34//! [token_span_end: u64]   — token span end
35//! [created_at: u64]       — creation timestamp (nanos)
36//! [updated_at: u64]       — last update timestamp (nanos)
37//! [importance: f32]       — importance score
38//! [tags: u32]             — tag bitfield
39//! [tier: u8]              — maturity tier enum
40//! [codec_id: u8]          — compression codec applied to key/value bytes (v2 only)
41//! [key_scales_len: u32]   — number of Q4 scale floats for key
42//! [key_data_len: u32]     — on-disk byte length of the (codec-encoded) key payload
43//! [val_scales_len: u32]   — number of Q4 scale floats for value
44//! [val_data_len: u32]     — on-disk byte length of the (codec-encoded) value payload
45//! [key_scales: ...]       — f32 scale factors
46//! [key_data: ...]         — codec-encoded Q4 bytes
47//! [val_scales: ...]       — f32 scale factors
48//! [val_data: ...]         — codec-encoded Q4 bytes
49//! [pos_encoding: ...]     — f32 values (unquantized)
50//! ```
51//!
52//! In v1 records the `codec_id` byte is absent; everything else has the same
53//! layout, and the key/value payloads are uncompressed Q4 bytes.

66/// File header magic bytes to identify segment files.
67const SEGMENT_MAGIC: &[u8; 4] = b"TDBS";
68/// Legacy segment format — no per-record codec byte. Still readable.
69const SEGMENT_VERSION_V1: u32 = 1;
70/// Current segment format — adds a per-record `codec_id` byte so cells can be
71/// encoded by different codecs in the same segment.
72const SEGMENT_VERSION_V2: u32 = 2;
73/// Size of the file header: magic(4) + version(4) = 8 bytes.
74const FILE_HEADER_SIZE: u64 = 8;
75
76/// Select the on-disk compression codec for a cell based on its maturity tier.
77///
78/// Draft cells turn over fast and use the raw Q4 byte stream so writes pay no
79/// codec cost. Validated and Core cells are warm-but-stable and earn the
80/// ~3–4× footprint shrink from zstd.
81fn codec_for_tier(tier: Tier) -> CompressionCodec {
82    match tier {
83        Tier::Draft => CompressionCodec::UniformQ4,
84        Tier::Validated | Tier::Core => CompressionCodec::ZstdQ4,
85    }
86}

95/// A single segment file — append-only storage for serialized memory cells.
96#[derive(Debug)]
97pub struct Segment {
98    id: u32,
99    path: PathBuf,
100    size: u64,
101    /// On-disk format version (read from the file header on `open`, set to the
102    /// current version on `create`). Drives the dispatch in [`Self::read_at`].
103    version: u32,
104}

106impl Segment {
107    /// Create a new empty segment file.
108    ///
109    /// # Errors
110    /// Returns an [`io::Error`] if the segment file cannot be created or
111    /// the magic/version header cannot be written.
112    pub fn create(dir: &Path, id: u32) -> io::Result<Self> {
113        let path = segment_path(dir, id);
114        let mut file = File::create(&path)?;
115        file.write_all(SEGMENT_MAGIC)?;
116        file.write_all(&SEGMENT_VERSION_V2.to_le_bytes())?;
117        file.flush()?;
118        Ok(Self { id, path, size: FILE_HEADER_SIZE, version: SEGMENT_VERSION_V2 })
119    }
120
121    /// Open an existing segment file.
122    ///
123    /// Reads the file header to discover the on-disk format version, which the
124    /// read path uses to decide whether each record carries a `codec_id` byte.
125    ///
126    /// # Errors
127    /// Returns an [`io::Error`] if the segment file does not exist, its
128    /// metadata cannot be read, its magic header is wrong, or its version is
129    /// not one the current binary understands.
130    pub fn open(dir: &Path, id: u32) -> io::Result<Self> {
131        let path = segment_path(dir, id);
132        let meta = fs::metadata(&path)?;
133        let size = meta.len();
134        let version = if size >= FILE_HEADER_SIZE {
135            let mut file = File::open(&path)?;
136            let mut magic = [0u8; 4];
137            file.read_exact(&mut magic)?;
138            if &magic != SEGMENT_MAGIC {
139                return Err(io::Error::new(
140                    io::ErrorKind::InvalidData,
141                    format!("invalid segment magic in {}", path.display()),
142                ));
143            }
144            let v = read_u32(&mut file)?;
145            if v != SEGMENT_VERSION_V1 && v != SEGMENT_VERSION_V2 {
146                return Err(io::Error::new(
147                    io::ErrorKind::InvalidData,
148                    format!("segment {} has unsupported format version {v}", path.display()),
149                ));
150            }
151            v
152        } else {
153            // Empty / header-truncated file — treat as the current version so the
154            // first append produces a well-formed segment.
155            SEGMENT_VERSION_V2
156        };
157        Ok(Self { id, path, size, version })
158    }

176    /// Append a `MemoryCell` (quantized to Q4) and return its byte offset.
177    ///
178    /// # Errors
179    /// Returns an [`io::Error`] if the segment file cannot be opened in
180    /// append mode, the record cannot be written, or fsync fails.
181    pub fn append(&mut self, cell: &MemoryCell) -> io::Result<u64> {
182        let file = OpenOptions::new().append(true).open(&self.path)?;
183        let mut w = BufWriter::new(file);
184
185        let offset = self.size;
186        let record_bytes = write_cell_record(&mut w, cell)?;
187        self.size = offset + 4 + record_bytes as u64;
188
189        let inner = w.into_inner().map_err(std::io::IntoInnerError::into_error)?;
190        inner.sync_data()?;
191        Ok(offset)
192    }
193
194    /// Append multiple cells in a single write + single fsync (Write-Behind Buffer).
195    ///
196    /// Returns a `Vec` of (`byte_offset`, `record_size`) for each cell written.
197    /// All cells are durably committed after the single `sync_data()` call.
198    ///
199    /// # Errors
200    /// Same as [`Self::append`]: open/write/fsync failures are surfaced as [`io::Error`].
201    pub fn append_batch(&mut self, cells: &[MemoryCell]) -> io::Result<Vec<u64>> {
202        if cells.is_empty() {
203            return Ok(Vec::new());
204        }
205
206        let file = OpenOptions::new().append(true).open(&self.path)?;
207        let mut w = BufWriter::new(file);
208
209        let mut offsets = Vec::with_capacity(cells.len());
210        for cell in cells {
211            let offset = self.size;
212            let record_bytes = write_cell_record(&mut w, cell)?;
213            self.size = offset + 4 + record_bytes as u64;
214            offsets.push(offset);
215        }
216
217        // Single fsync for all cells — this is the key optimization.
218        let inner = w.into_inner().map_err(std::io::IntoInnerError::into_error)?;
219        inner.sync_data()?;
220        Ok(offsets)
221    }

223    /// Read a `MemoryCell` from a specific byte offset.
224    ///
225    /// # Errors
226    /// Returns an [`io::Error`] if the segment cannot be opened, the seek
227    /// fails, any field-by-field read short-reads, the tier byte is not
228    /// one of `0`/`1`/`2` ([`io::ErrorKind::InvalidData`]), or a v2 record
229    /// carries an unknown `codec_id`.
230    pub fn read_at(&self, byte_offset: u64) -> io::Result<MemoryCell> {
231        let mut file = File::open(&self.path)?;
232        file.seek(SeekFrom::Start(byte_offset))?;
233
234        let record_len = read_u32(&mut file)?;
235        let _ = record_len; // We read field-by-field, not by total length.
236
237        let cell_id: CellId = read_u64(&mut file)?;
238        let owner: OwnerId = read_u64(&mut file)?;
239        let layer: LayerId = read_u16(&mut file)?;
240        let key_dim = read_u32(&mut file)? as usize;
241        let value_dim = read_u32(&mut file)? as usize;
242        let pos_dim = read_u32(&mut file)? as usize;
243        let token_span_start = read_u64(&mut file)?;
244        let token_span_end = read_u64(&mut file)?;
245        let created_at = read_u64(&mut file)?;
246        let updated_at = read_u64(&mut file)?;
247        let importance = read_f32(&mut file)?;
248        let tags = read_u32(&mut file)?;
249        let tier_byte = read_u8(&mut file)?;
250
251        // v2 adds a codec_id byte right after tier. v1 segments wrote nothing
252        // here and assume `UniformQ4` (legacy Q4-on-disk).
253        let codec = if self.version >= SEGMENT_VERSION_V2 {
254            let codec_byte = read_u8(&mut file)?;
255            CompressionCodec::from_u8(codec_byte)?
256        } else {
257            CompressionCodec::UniformQ4
258        };
259
260        let key_scales_len = read_u32(&mut file)? as usize;
261        let key_data_len = read_u32(&mut file)? as usize;
262        let val_scales_len = read_u32(&mut file)? as usize;
263        let val_data_len = read_u32(&mut file)? as usize;
264
265        let key_scales = read_f32_vec(&mut file, key_scales_len)?;
266        let key_on_disk = read_bytes(&mut file, key_data_len)?;
267        let val_scales = read_f32_vec(&mut file, val_scales_len)?;
268        let val_on_disk = read_bytes(&mut file, val_data_len)?;
269        let pos_encoding = read_f32_vec(&mut file, pos_dim)?;
270
271        // Decode bytes back to the raw Q4 stream the dequantizer understands.
272        let key_data = codec.decode(&key_on_disk)?;
273        let val_data = codec.decode(&val_on_disk)?;
274
275        let key_q = QuantizedTensor { data: key_data, scales: key_scales, original_len: key_dim };
276        let val_q = QuantizedTensor { data: val_data, scales: val_scales, original_len: value_dim };
277
278        let key = Q4::dequantize(&key_q);
279        let value = Q4::dequantize(&val_q);
280
281        let tier = match tier_byte {
282            0 => Tier::Draft,
283            1 => Tier::Validated,
284            2 => Tier::Core,
285            _ => {
286                return Err(io::Error::new(
287                    io::ErrorKind::InvalidData,
288                    format!("invalid tier byte {tier_byte} in cell {cell_id}"),
289                ));
290            }
291        };
292
293        Ok(MemoryCell {
294            id: cell_id,
295            owner,
296            layer,
297            key,
298            value,
299            token_span: (token_span_start, token_span_end),
300            pos_encoding,
301            meta: CellMeta { created_at, updated_at, importance, tags, tier },
302        })
303    }
304}

306/// Scan a segment to rebuild the index (used on recovery).
307///
308/// # Errors
309/// Returns an [`io::Error`] if the segment file cannot be opened or its
310/// magic header does not match the expected `SEGMENT_MAGIC`
311/// ([`io::ErrorKind::InvalidData`]). Truncated trailing records are
312/// silently discarded.
313pub fn scan_segment(dir: &Path, segment_id: u32) -> io::Result<Vec<(CellId, u64)>> {
314    let path = segment_path(dir, segment_id);
315    let mut file = File::open(&path)?;
316    let file_len = file.metadata()?.len();
317
318    // Validate file header.
319    if file_len < FILE_HEADER_SIZE {
320        return Ok(Vec::new());
321    }
322    let mut magic = [0u8; 4];
323    file.read_exact(&mut magic)?;
324    if &magic != SEGMENT_MAGIC {
325        return Err(io::Error::new(
326            io::ErrorKind::InvalidData,
327            format!("invalid segment magic in {}", path.display()),
328        ));
329    }
330    // Skip version (4 bytes) — already past magic.
331    file.seek(SeekFrom::Start(FILE_HEADER_SIZE))?;
332
333    let mut entries = Vec::new();
334    let mut pos = FILE_HEADER_SIZE;
335
336    while pos < file_len {
337        let record_len = match read_u32(&mut file) {
338            Ok(len) => u64::from(len),
339            Err(_) => break, // truncated record_len — stop scanning
340        };
341
342        // Bounds check: ensure the full record fits within the file.
343        if pos + 4 + record_len > file_len {
344            break; // partial record at EOF — discard it
345        }
346
347        let Ok(cell_id) = read_u64(&mut file) else {
348            break; // partial record — stop scanning
349        };
350        entries.push((cell_id, pos));
351
352        // Skip the rest of this record.
353        let remaining = record_len.saturating_sub(8);
354        let Ok(seek_offset) = i64::try_from(remaining) else {
355            break;
356        };
357        if file.seek(SeekFrom::Current(seek_offset)).is_err() {
358            break;
359        }
360        pos += 4 + record_len;
361    }
362
363    Ok(entries)
364}

370/// List segment IDs found in a directory, sorted.
371///
372/// # Errors
373/// Returns an [`io::Error`] if the directory cannot be read.
374pub fn list_segments(dir: &Path) -> io::Result<Vec<u32>> {
375    let mut ids = Vec::new();
376    for entry in fs::read_dir(dir)? {
377        let entry = entry?;
378        let name = entry.file_name();
379        let name = name.to_string_lossy();
380        if let Some(rest) = name.strip_prefix("segment_")
381            && let Some(num_str) = rest.strip_suffix(".tdb")
382            && let Ok(id) = num_str.parse::<u32>()
383        {
384            ids.push(id);
385        }
386    }
387    ids.sort_unstable();
388    Ok(ids)
389}

391/// Compute the v2 record body size given the on-disk (codec-encoded) payload sizes.
392fn compute_record_size_v2(
393    key_scales_count: usize,
394    key_payload_bytes: usize,
395    val_scales_count: usize,
396    val_payload_bytes: usize,
397    pos_len: usize,
398) -> usize {
399    // Fixed v2 fields: id(8) + owner(8) + layer(2) + key_dim(4) + value_dim(4) + pos_dim(4)
400    //   + token_span(16) + created_at(8) + updated_at(8) + importance(4) + tags(4) + tier(1)
401    //   + codec_id(1) + key_scales_len(4) + key_data_len(4) + val_scales_len(4) + val_data_len(4)
402    let fixed = 8 + 8 + 2 + 4 + 4 + 4 + 16 + 8 + 8 + 4 + 4 + 1 + 1 + 4 + 4 + 4 + 4;
403    let variable = key_scales_count * 4
404        + key_payload_bytes
405        + val_scales_count * 4
406        + val_payload_bytes
407        + pos_len * 4;
408    fixed + variable
409}

413/// Serialize a single `MemoryCell` to a writer using the v2 record layout
414/// (with a `codec_id` byte) and the tier-selected compression codec.
415///
416/// Returns the record body size (excluding the 4-byte `record_len` prefix).
417fn write_cell_record(w: &mut impl Write, cell: &MemoryCell) -> io::Result<u64> {
418    let codec = codec_for_tier(cell.meta.tier);
419    let key_q = Q4::quantize(&cell.key);
420    let val_q = Q4::quantize(&cell.value);
421
422    // Encode the Q4 byte streams through the selected codec *before* sizing
423    // the record so the on-disk byte counts are known up front.
424    let key_payload = codec.encode(&key_q.data)?;
425    let val_payload = codec.encode(&val_q.data)?;
426
427    let record_bytes = compute_record_size_v2(
428        key_q.scales.len(),
429        key_payload.len(),
430        val_q.scales.len(),
431        val_payload.len(),
432        cell.pos_encoding.len(),
433    );
434    let record_len: u32 = record_bytes.try_into().map_err(|_| {
435        io::Error::new(
436            io::ErrorKind::InvalidInput,
437            format!("record size {record_bytes} exceeds u32::MAX"),
438        )
439    })?;
440    w.write_all(&record_len.to_le_bytes())?;
441
442    // Reason for the `as u32` casts below: every length is a component of
443    // `record_bytes`, which the `try_into::<u32>` guard above just verified
444    // fits in u32. Each component is therefore also ≤ u32::MAX. The acceptance
445    // test `acceptance::record_overflow_returns_invalid_input` exercises the
446    // outer guard.
447    let key_len = u32_from_len(cell.key.len());
448    let value_len = u32_from_len(cell.value.len());
449    let pos_len = u32_from_len(cell.pos_encoding.len());
450    let key_scales_len = u32_from_len(key_q.scales.len());
451    let key_data_len = u32_from_len(key_payload.len());
452    let val_scales_len = u32_from_len(val_q.scales.len());
453    let val_data_len = u32_from_len(val_payload.len());
454
455    // Fixed fields.
456    w.write_all(&cell.id.to_le_bytes())?;
457    w.write_all(&cell.owner.to_le_bytes())?;
458    w.write_all(&cell.layer.to_le_bytes())?;
459    w.write_all(&key_len.to_le_bytes())?;
460    w.write_all(&value_len.to_le_bytes())?;
461    w.write_all(&pos_len.to_le_bytes())?;
462    w.write_all(&cell.token_span.0.to_le_bytes())?;
463    w.write_all(&cell.token_span.1.to_le_bytes())?;
464    w.write_all(&cell.meta.created_at.to_le_bytes())?;
465    w.write_all(&cell.meta.updated_at.to_le_bytes())?;
466    w.write_all(&cell.meta.importance.to_le_bytes())?;
467    w.write_all(&cell.meta.tags.to_le_bytes())?;
468    w.write_all(&[cell.meta.tier as u8])?;
469    w.write_all(&[codec.as_u8()])?;
470
471    // Quantized key.
472    w.write_all(&key_scales_len.to_le_bytes())?;
473    w.write_all(&key_data_len.to_le_bytes())?;
474    // Quantized value.
475    w.write_all(&val_scales_len.to_le_bytes())?;
476    w.write_all(&val_data_len.to_le_bytes())?;
477
478    // Variable-length data.
479    write_f32_slice(w, &key_q.scales)?;
480    w.write_all(&key_payload)?;
481    write_f32_slice(w, &val_q.scales)?;
482    w.write_all(&val_payload)?;
483    write_f32_slice(w, &cell.pos_encoding)?;
484
485    Ok(record_bytes as u64)
486}

488/// Saturating-to-u32 length helper for serialization.
489///
490/// Used by [`write_cell_record`] where the caller has already validated that
491/// the *total* record size fits in u32; each component length therefore also
492/// fits in u32. The saturating cast preserves the existing behavior — the
493/// outer guard would have rejected the cell before any of these casts run.
494#[inline]
495fn u32_from_len(len: usize) -> u32 {
496    u32::try_from(len).unwrap_or(u32::MAX)
497}

538fn read_f32_vec(r: &mut impl Read, count: usize) -> io::Result<Vec<f32>> {
539    let mut buf = vec![0u8; count * 4];
540    r.read_exact(&mut buf)?;
541    Ok(buf.chunks_exact(4).map(|c| f32::from_le_bytes([c[0], c[1], c[2], c[3]])).collect())
542}
