1//! Durability boundary tracking — the storage-layer contract that
2//! lets engine consumers reason about "what is durable" without
3//! coupling to the concrete tracker implementation.
4//!
5//! # Why this lives in `tdb-storage`
6//!
7//! Durability is fundamentally a storage concern: the offset advances
8//! when a segment fsync returns. Putting the abstraction in
9//! `tdb-storage` lets the storage layer own its own contract instead
10//! of expressing it as a concrete struct nested inside the engine.
11//! Future replicated / async-batched durability impls can replace
12//! [`DurabilityTracker`] under the trait without touching the engine.
13//!
14//! # The contract
15//!
16//! Two monotonic counters express the durability boundary:
17//!
18//! - `issued`: bumps every time a write is accepted (regardless of
19//!   whether the underlying storage has fsynced it). A write goes
20//!   from "issued" to "durable" via an fsync somewhere downstream.
21//! - `durable`: bumps every time an fsync completes successfully.
22//!   Always `<= issued`. Catches up to `issued` after each flush.
23//!
24//! Confirmed reads snapshot `issued` at request entry, run the
25//! retrieval pipeline, then wait until `durable` reaches the
26//! snapshot — every concurrent in-flight write that existed at
27//! request time is now durable when the read returns.
28//!
29//! # Implementation
30//!
31//! [`DurabilityTracker`] uses `std::sync::Condvar` + `AtomicU64`, no
32//! async runtime. The project's existing concurrency model is
33//! `Arc<RwLock<>>` from std accessed under `py.detach()`; staying
34//! inside std avoids dragging tokio into a synchronous codebase
35//! just for one watch channel.
36//!
37//! # Invariants
38//!
39//! - `issued` and `durable` are both monotonically non-decreasing.
40//! - `durable <= issued` at all times.
41//! - `publish_durable(n)` is idempotent: republishing an offset at
42//!   or below the current value is a no-op.
43
44use std::sync::atomic::{AtomicU64, Ordering};
45use std::sync::{Condvar, Mutex};
46use std::time::{Duration, Instant};

48/// Error returned by [`Durability::wait_durable`] when the caller's
49/// deadline passes before `durable` reaches the target.
50#[derive(Debug, Clone, Copy)]
51pub struct WaitTimedOut {
52    /// Elapsed wall-clock time the caller spent waiting before
53    /// giving up. Useful for surfacing in error messages.
54    pub waited: Duration,
55}

57/// The durability boundary contract.
58///
59/// Every concrete durability impl provides the same four observable
60/// facts: the current durable offset, the current issued offset, a
61/// way to bump issued, and a way to block until durable catches up
62/// to a target. Engine consumers depend on this trait so a future
63/// replicated / async-batched / metric-instrumented impl can plug in
64/// without touching the engine's write or read paths.
65///
66/// Object-safe by construction — methods are `&self`, no generics,
67/// no `Self` return types. Stored in the engine as
68/// `Arc<dyn Durability + Send + Sync>`.
69pub trait Durability: std::fmt::Debug + Send + Sync {
70    /// Current durable offset — every write at or below this offset
71    /// has reached stable storage.
72    fn current_durable(&self) -> u64;
73
74    /// Current issued offset — every accepted write up to and
75    /// including this offset is in flight or durable.
76    fn current_issued(&self) -> u64;
77
78    /// Atomically bump `issued` and return the new value.
79    ///
80    /// Called from every write path the moment the engine accepts
81    /// the write. Distinct from durability — the write may sit in a
82    /// buffer for an arbitrary time before its fsync lands.
83    fn issue_one(&self) -> u64;
84
85    /// Mark every issued offset up to and including `offset` as
86    /// durable. Idempotent: a second call with the same or lower
87    /// value is a no-op.
88    ///
89    /// Implementations **must** clamp `offset` to `current_issued()`
90    /// so `durable <= issued` is an unconditional invariant.
91    fn publish_durable(&self, offset: u64);
92
93    /// Block until `current_durable() >= target` or `deadline`
94    /// elapses.
95    ///
96    /// Callers should drop the engine lock before invoking this so
97    /// writers can advance the offset concurrently.
98    ///
99    /// # Errors
100    ///
101    /// Returns [`WaitTimedOut`] if `current_durable()` does not reach
102    /// `target` within `deadline`.
103    fn wait_durable(&self, target: u64, deadline: Duration) -> Result<(), WaitTimedOut>;
104}

106/// In-process durability tracker — the default [`Durability`]
107/// implementation backing every embedded engine instance.
108///
109/// One instance per engine. Cloneable references via [`std::sync::Arc`]
110/// let readers wait without holding the engine lock — the Python
111/// binding's confirmed-read path borrows an `Arc<dyn Durability>`,
112/// drops the engine read lock, then calls [`Self::wait_durable`]
113/// outside the lock so writers can advance the offset concurrently.
114#[derive(Debug, Default)]
115pub struct DurabilityTracker {
116    issued: AtomicU64,
117    durable: AtomicU64,
118    /// Companion to [`Self::notify`]. The Condvar protocol requires
119    /// holding this mutex when both signalling and waiting.
120    notify_lock: Mutex<()>,
121    notify: Condvar,
122}
123
124impl DurabilityTracker {
125    /// Construct an empty tracker. Both offsets start at 0.
126    #[must_use]
127    pub fn new() -> Self {
128        Self {
129            issued: AtomicU64::new(0),
130            durable: AtomicU64::new(0),
131            notify_lock: Mutex::new(()),
132            notify: Condvar::new(),
133        }
134    }
135}
136
137impl Durability for DurabilityTracker {
138    fn current_durable(&self) -> u64 {
139        self.durable.load(Ordering::Acquire)
140    }
141
142    fn current_issued(&self) -> u64 {
143        self.issued.load(Ordering::Acquire)
144    }
145
146    fn issue_one(&self) -> u64 {
147        self.issued.fetch_add(1, Ordering::AcqRel) + 1
148    }
149
150    fn publish_durable(&self, offset: u64) {
151        // Clamp to issued — the durability boundary cannot lead
152        // acceptance. Caught by a property test in
153        // `tests/durability_proptest.rs::durable_never_exceeds_issued`.
154        let effective = offset.min(self.issued.load(Ordering::Acquire));
155        // `fetch_max` semantics via CAS — keep the maximum so
156        // out-of-order publishes don't regress the offset.
157        let mut current = self.durable.load(Ordering::Acquire);
158        loop {
159            if effective <= current {
160                return;
161            }
162            match self.durable.compare_exchange_weak(
163                current,
164                effective,
165                Ordering::AcqRel,
166                Ordering::Acquire,
167            ) {
168                Ok(_) => break,
169                Err(actual) => current = actual,
170            }
171        }
172        // Acquiring the notify mutex here means waiters that
173        // observed `durable < target` and are about to wait have
174        // either already entered `wait_timeout` (and will be woken
175        // by `notify_all`) or are still holding the mutex (in which
176        // case we serialise behind them and they see the updated
177        // value on their next check).
178        let _guard = self.notify_lock.lock().expect("durability notify mutex poisoned");
179        self.notify.notify_all();
180    }
181
182    fn wait_durable(&self, target: u64, deadline: Duration) -> Result<(), WaitTimedOut> {
183        // Fast path — already durable, skip the mutex acquire.
184        if self.current_durable() >= target {
185            return Ok(());
186        }
187        let start = Instant::now();
188        let mut guard = self.notify_lock.lock().expect("durability notify mutex poisoned");
189        loop {
190            if self.current_durable() >= target {
191                return Ok(());
192            }
193            let elapsed = start.elapsed();
194            // `checked_sub` rather than `-`: clippy's strict mode
195            // forbids the unguarded operator on `Duration`.
196            let Some(remaining) = deadline.checked_sub(elapsed) else {
197                return Err(WaitTimedOut { waited: elapsed });
198            };
199            let (next_guard, timeout_result) = self
200                .notify
201                .wait_timeout(guard, remaining)
202                .expect("durability notify mutex poisoned");
203            guard = next_guard;
204            if timeout_result.timed_out() && self.current_durable() < target {
205                return Err(WaitTimedOut { waited: start.elapsed() });
206            }
207        }
208    }
209}

211#[cfg(test)]
212mod tests {
213    use super::*;
214    use std::sync::Arc;
215    use std::thread;
216
217    #[test]
218    fn new_tracker_has_zero_offsets() {
219        let t = DurabilityTracker::new();
220        assert_eq!(t.current_durable(), 0);
221        assert_eq!(t.current_issued(), 0);
222    }
223
224    #[test]
225    fn issue_one_returns_strictly_increasing_offsets() {
226        let t = DurabilityTracker::new();
227        assert_eq!(t.issue_one(), 1);
228        assert_eq!(t.issue_one(), 2);
229        assert_eq!(t.issue_one(), 3);
230        assert_eq!(t.current_issued(), 3);
231    }
232
233    /// Helper: bump `issued` to a target value so the tracker's
234    /// invariant `durable <= issued` permits a publish at that
235    /// level. Mirrors the engine's own write-then-fsync pattern.
236    fn issue_to(t: &DurabilityTracker, target: u64) {
237        while t.current_issued() < target {
238            t.issue_one();
239        }
240    }
241
242    #[test]
243    fn publish_durable_is_monotonic() {
244        let t = DurabilityTracker::new();
245        issue_to(&t, 10);
246        t.publish_durable(5);
247        assert_eq!(t.current_durable(), 5);
248        t.publish_durable(3); // Lower — must not regress.
249        assert_eq!(t.current_durable(), 5);
250        t.publish_durable(10);
251        assert_eq!(t.current_durable(), 10);
252    }
253
254    #[test]
255    fn wait_durable_returns_immediately_when_already_at_target() {
256        let t = DurabilityTracker::new();
257        issue_to(&t, 5);
258        t.publish_durable(5);
259        let start = Instant::now();
260        assert!(t.wait_durable(3, Duration::from_mins(1)).is_ok());
261        assert!(start.elapsed() < Duration::from_millis(10));
262    }
263
264    #[test]
265    fn publish_durable_clamps_to_issued() {
266        // Documented invariant: publishing past `issued` advances
267        // durable only as far as `issued` permits. Caught by the
268        // proptest gate; pinned here as an example.
269        let t = DurabilityTracker::new();
270        t.publish_durable(100);
271        assert_eq!(t.current_durable(), 0, "no writes issued — durable must stay at 0");
272        t.issue_one();
273        t.publish_durable(100);
274        assert_eq!(t.current_durable(), 1, "one write issued — durable clamps to 1");
275    }
276
277    #[test]
278    fn wait_durable_times_out_when_no_publish_occurs() {
279        let t = DurabilityTracker::new();
280        let err = t.wait_durable(1, Duration::from_millis(50)).unwrap_err();
281        assert!(err.waited >= Duration::from_millis(50));
282        assert!(err.waited < Duration::from_millis(500));
283    }
284
285    #[test]
286    fn wait_durable_unblocks_when_publish_arrives() {
287        let t = Arc::new(DurabilityTracker::new());
288        // Issue ahead so publish can advance durable to 5.
289        issue_to(&t, 5);
290        let waiter = {
291            let t = Arc::clone(&t);
292            thread::spawn(move || t.wait_durable(5, Duration::from_secs(2)))
293        };
294        // Give the waiter a moment to enter its wait.
295        thread::sleep(Duration::from_millis(50));
296        t.publish_durable(5);
297        let result = waiter.join().unwrap();
298        assert!(result.is_ok());
299    }
300
301    #[test]
302    fn multiple_waiters_unblock_on_single_publish() {
303        let t = Arc::new(DurabilityTracker::new());
304        issue_to(&t, 7);
305        let waiters: Vec<_> = (0..3)
306            .map(|_| {
307                let t = Arc::clone(&t);
308                thread::spawn(move || t.wait_durable(7, Duration::from_secs(2)))
309            })
310            .collect();
311        thread::sleep(Duration::from_millis(50));
312        t.publish_durable(7);
313        for waiter in waiters {
314            assert!(waiter.join().unwrap().is_ok());
315        }
316    }
317
318    /// Verify the trait is dyn-compatible — `Arc<dyn Durability>`
319    /// must compile. If this stops working, callers that store the
320    /// tracker as `Arc<dyn Durability + Send + Sync>` will break.
321    #[test]
322    fn durability_trait_is_object_safe() {
323        let tracker: Arc<dyn Durability + Send + Sync> = Arc::new(DurabilityTracker::new());
324        assert_eq!(tracker.current_durable(), 0);
325        let issued = tracker.issue_one();
326        tracker.publish_durable(issued);
327        assert_eq!(tracker.current_durable(), issued);
328    }
