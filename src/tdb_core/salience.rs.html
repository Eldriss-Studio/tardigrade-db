1//! Salience derivation modes for pack writes.
2//!
3//! A pack's *salience* is a 0..100 score that seeds its initial position
4//! on the importance ladder. Two write contracts coexist:
5//!
6//! * The caller may pass an explicit salience the engine stores verbatim
7//!   (the default for legacy `mem_write_pack` callers).
8//! * The caller may pass a [`SalienceMode`] and let the engine derive
9//!   salience from the retrieval key's magnitude. This keeps the formula
10//!   in Rust, out of every consumer's Python layer, and consistent across
11//!   producers (`HuggingFace` hook, vLLM connector, MCP server).
12//!
13//! All derived modes apply [`SALIENCE_SCALE`] and clamp at [`SALIENCE_CAP`]
14//! so callers cannot accidentally over-rate a pack on a single write.
15
16/// Multiplier mapping a per-element retrieval-key magnitude into the
17/// 0..100 salience range. Chosen so that typical N(0, 1) hidden-state
18/// data sits in the 5..40 band — well clear of the cap.
19pub const SALIENCE_SCALE: f32 = 50.0;
20
21/// Maximum salience an automated derivation may assign. Caller-supplied
22/// salience is **not** clamped — explicit always wins.
23pub const SALIENCE_CAP: f32 = 100.0;
24
25/// How the engine should derive salience from the encoded retrieval key.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum SalienceMode {
28    /// Preserve the caller's explicit salience. The default — and the
29    /// behaviour when no `salience_mode` argument is passed.
30    Explicit,
31    /// `min(SALIENCE_SCALE * (||key_data|| / len(key_data)), SALIENCE_CAP)`.
32    /// L2 norm of the *non-header* portion of the encoded key, divided
33    /// by the number of elements to get a per-element magnitude.
34    L2,
35    /// `min(SALIENCE_SCALE * max(|key_data|), SALIENCE_CAP)`. Peak
36    /// magnitude rather than average — surfaces packs with at least one
37    /// strong activation even if most tokens are flat.
38    Max,
39}
40
41impl SalienceMode {
42    /// Parse the string form callers pass through the `PyO3` layer.
43    ///
44    /// # Errors
45    /// Returns [`UnknownSalienceMode`] if `name` is not one of
46    /// `"none"`, `"explicit"`, `"l2"`, or `"max"`.
47    pub fn parse(name: &str) -> Result<Self, UnknownSalienceMode> {
48        match name {
49            "none" | "explicit" => Ok(Self::Explicit),
50            "l2" => Ok(Self::L2),
51            "max" => Ok(Self::Max),
52            other => Err(UnknownSalienceMode { name: other.to_owned() }),
53        }
54    }
55
56    /// Resolve the effective salience given the caller's explicit value
57    /// and the encoded retrieval key. The `header_size` argument is how
58    /// many leading f32s of `encoded_key` to skip before measuring —
59    /// callers pass `tdb_retrieval::per_token::HEADER_SIZE` for the
60    /// per-token encoded form.
61    pub fn resolve(self, explicit: f32, encoded_key: &[f32], header_size: usize) -> f32 {
62        if self == Self::Explicit || encoded_key.len() <= header_size {
63            return explicit;
64        }
65        let data = &encoded_key[header_size..];
66        let raw = match self {
67            Self::L2 => {
68                let n = data.len().max(1) as f32;
69                let sum_sq: f32 = data.iter().map(|v| v * v).sum();
70                (sum_sq.sqrt() / n) * SALIENCE_SCALE
71            }
72            Self::Max => {
73                let abs_max = data.iter().map(|v| v.abs()).fold(0.0_f32, f32::max);
74                abs_max * SALIENCE_SCALE
75            }
76            Self::Explicit => return explicit,
77        };
78        raw.min(SALIENCE_CAP)
79    }
80}
81
82/// Returned when [`SalienceMode::parse`] receives an unknown name.
83#[derive(Debug, Clone)]
84pub struct UnknownSalienceMode {
85    pub name: String,
86}
87
88impl std::fmt::Display for UnknownSalienceMode {
89    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
90        write!(
91            f,
92            "unknown salience_mode {:?}; expected one of \"l2\", \"max\", \"none\"",
93            self.name
94        )
95    }
96}
97
98impl std::error::Error for UnknownSalienceMode {}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    const HEADER: usize = 4;
105
106    #[test]
107    fn explicit_returns_caller_value() {
108        assert!((SalienceMode::Explicit.resolve(42.0, &[0.0; 32], HEADER) - 42.0).abs() < 1e-6);
109    }
110
111    #[test]
112    fn l2_uses_non_header_portion() {
113        // Header is all zeros; data is [1.0, 1.0, 1.0, 1.0].
114        let mut key = vec![0.0_f32; HEADER];
115        key.extend([1.0; 4]);
116        let s = SalienceMode::L2.resolve(0.0, &key, HEADER);
117        // L2 = sqrt(4) = 2.0; per-element = 0.5; scaled = 25.0.
118        assert!((s - 25.0).abs() < 1e-4, "l2 salience {s} != 25.0");
119    }
120
121    #[test]
122    fn max_uses_inf_norm_of_non_header_portion() {
123        let mut key = vec![0.0_f32; HEADER];
124        key.extend([0.1, 0.5, -0.7, 0.3]);
125        let s = SalienceMode::Max.resolve(0.0, &key, HEADER);
126        // max(|x|) = 0.7; scaled = 35.0.
127        assert!((s - 35.0).abs() < 1e-4, "max salience {s} != 35.0");
128    }
129
130    #[test]
131    fn clamps_at_cap() {
132        let mut key = vec![0.0_f32; HEADER];
133        key.extend([100.0_f32; 4]);
134        assert!((SalienceMode::Max.resolve(0.0, &key, HEADER) - SALIENCE_CAP).abs() < 1e-6);
135    }
136
137    #[test]
138    fn short_key_falls_back_to_explicit() {
139        let key = vec![0.0_f32; HEADER];
140        assert!((SalienceMode::L2.resolve(7.5, &key, HEADER) - 7.5).abs() < 1e-6);
141    }
142
143    #[test]
144    fn parse_accepts_known_modes() {
145        assert_eq!(SalienceMode::parse("l2").unwrap(), SalienceMode::L2);
146        assert_eq!(SalienceMode::parse("max").unwrap(), SalienceMode::Max);
147        assert_eq!(SalienceMode::parse("none").unwrap(), SalienceMode::Explicit);
148        assert_eq!(SalienceMode::parse("explicit").unwrap(), SalienceMode::Explicit);
149    }
150
151    #[test]
152    fn parse_rejects_unknown_mode() {
153        assert!(SalienceMode::parse("banana").is_err());
154    }
155}
