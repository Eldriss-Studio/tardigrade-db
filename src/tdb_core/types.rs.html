16/// Maturity tier in the Adaptive Knowledge Lifecycle.
17///
18/// Transitions use hysteresis to prevent oscillation:
19/// - Draft → Validated at ι ≥ 65, demotes back at ι < 35
20/// - Validated → Core at ι ≥ 85, demotes back at ι < 60
21#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Hash)]
22#[repr(u8)]
23pub enum Tier {
24    #[default]
25    Draft = 0,
26    Validated = 1,
27    Core = 2,
28}
29
30/// Retrieval boost for Draft-tier memories (no advantage).
31const DRAFT_RETRIEVAL_BOOST: f32 = 1.0;
32/// Retrieval boost for Validated-tier memories (accessed enough to cross ι≥65).
33const VALIDATED_RETRIEVAL_BOOST: f32 = 1.1;
34/// Retrieval boost for Core-tier memories (stable, repeatedly accessed, ι≥85).
35const CORE_RETRIEVAL_BOOST: f32 = 1.25;
36
37impl Tier {
38    /// Score multiplier applied during retrieval based on maturity.
39    ///
40    /// Core memories have proven their value through repeated access;
41    /// they rank higher than untested Draft memories.
42    #[must_use]
43    pub fn retrieval_boost(self) -> f32 {
44        match self {
45            Self::Draft => DRAFT_RETRIEVAL_BOOST,
46            Self::Validated => VALIDATED_RETRIEVAL_BOOST,
47            Self::Core => CORE_RETRIEVAL_BOOST,
48        }
49    }
50}

52/// Read-visibility mode for retrieval APIs that participate in the
53/// confirmed-vs-unconfirmed durability contract.
54///
55/// [`ReadVisibility::Unconfirmed`] is the default and existing
56/// behaviour: the read returns immediately with whatever the
57/// retrieval pipeline currently has. [`ReadVisibility::Confirmed`]
58/// blocks until every write issued before the read became durable.
59///
60/// See CLAUDE.md's "Reliability & Consistency Rules" section for the
61/// project-canonical contract this enum surfaces, and
62/// `~/.claude/plans/spacetimedb-confirmed-reads-contract.md` for the
63/// implementation plan.
64///
65/// # Why an explicit timeout is required on `Confirmed`
66///
67/// A confirmed read with no deadline can block forever — if the
68/// underlying write fails fsync without surfacing an error, the
69/// reader's wait will never complete. Requiring an explicit deadline
70/// makes that failure path bounded: the reader sees a `ReadTimeout`
71/// instead of hanging, and the consumer's higher-level error
72/// handling can decide what to do.
73#[derive(Debug, Clone, Copy, Default)]
74pub enum ReadVisibility {
75    /// Default mode — return immediately, no durability wait.
76    #[default]
77    Unconfirmed,
78    /// Block until the snapshot offset captured at request entry is
79    /// durable, or until `timeout` elapses (whichever comes first).
80    Confirmed {
81        /// Maximum wall-clock wait before returning `ReadTimeout`.
82        timeout: std::time::Duration,
83    },
84}
